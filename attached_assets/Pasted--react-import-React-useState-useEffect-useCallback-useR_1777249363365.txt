const WordLow = ({ onExit }) => {
  const BUZZWORDS = ['GHOST', 'TOXIC', 'CLOUT', 'VIBES', 'FLAKE', 'MATCH', 'SHADE', 'BREAD', 'GASLY', 'CRUSH'];
  const [targetWord, setTargetWord] = useState('');
  const [guesses, setGuesses] = useState([]);
  const [currentGuess, setCurrentGuess] = useState('');
  const [gameState, setGameState] = useState('PLAYING'); // PLAYING, WON, LOST

  useEffect(() => {
    setTargetWord(BUZZWORDS[Math.floor(Math.random() * BUZZWORDS.length)]);
  }, []);

  const onKeyPress = (key) => {
    if (gameState !== 'PLAYING') return;
    if (key === 'ENTER') {
      if (currentGuess.length !== 5) return;
      const newGuesses = [...guesses, currentGuess];
      setGuesses(newGuesses);
      if (currentGuess === targetWord) setGameState('WON');
      else if (newGuesses.length >= 6) setGameState('LOST');
      setCurrentGuess('');
    } else if (key === 'DEL') {
      setCurrentGuess(prev => prev.slice(0, -1));
    } else if (currentGuess.length < 5 && /^[A-Z]$/.test(key)) {
      setCurrentGuess(prev => prev + key);
    }
  };

  const getLetterStyle = (letter, pos, guess) => {
    if (targetWord[pos] === letter) return 'bg-teal-500 text-black border-teal-400 shadow-[0_0_15px_rgba(20,184,166,0.5)]';
    if (targetWord.includes(letter)) return 'bg-amber-600 text-black border-amber-500';
    return 'bg-zinc-800 text-zinc-500 border-zinc-700';
  };

  return (
    <div className="absolute inset-0 bg-zinc-950 flex flex-col p-6 pt-12 animate-in fade-in overflow-hidden font-mono">
      {/* Smoky Overlay */}
      <div className="absolute inset-0 pointer-events-none opacity-20 bg-[radial-gradient(circle_at_50%_50%,rgba(255,255,255,0.1),transparent)] animate-pulse"></div>
      
      <div className="flex justify-between items-center mb-8 border-b border-zinc-800 pb-2">
        <h2 className="text-white font-black italic text-xl uppercase tracking-widest">Word-Low</h2>
        <div className="text-[10px] text-zinc-500 font-bold uppercase tracking-widest">Poser Check: ON</div>
      </div>

      {/* Grid */}
      <div className="flex-1 flex flex-col gap-2 mb-4 justify-center">
        {[...Array(6)].map((_, i) => {
          const guess = guesses[i] || (i === guesses.length ? currentGuess : '');
          const isSubmitted = i < guesses.length;
          return (
            <div key={i} className="flex justify-center gap-2">
              {[...Array(5)].map((_, j) => {
                const char = guess[j] || '';
                let style = 'bg-zinc-900 border-zinc-800 text-white';
                if (isSubmitted) style = getLetterStyle(char, j, guess);
                return (
                  <div key={j} className={`w-12 h-12 border-2 flex items-center justify-center text-2xl font-black rounded-md transition-all duration-500 ${style}`}>
                    {char}
                  </div>
                );
              })}
            </div>
          );
        })}
      </div>

      {/* Typewriter Keyboard */}
      <div className="grid gap-1 mt-auto">
        {['QWERTYUIOP', 'ASDFGHJKL', 'ZXCVBNM'].map((row, i) => (
          <div key={i} className="flex justify-center gap-1">
            {i === 2 && <button onClick={() => onKeyPress('ENTER')} className="px-2 h-10 bg-zinc-800 text-[10px] font-bold rounded hover:bg-zinc-700 active:scale-95">ENTER</button>}
            {row.split('').map(key => (
              <button key={key} onClick={() => onKeyPress(key)} className="w-7 h-10 bg-zinc-900 border border-zinc-800 text-white font-bold rounded hover:bg-zinc-700 active:scale-95 shadow-lg">
                {key}
              </button>
            ))}
            {i === 2 && <button onClick={() => onKeyPress('DEL')} className="px-2 h-10 bg-zinc-800 text-[10px] font-bold rounded hover:bg-zinc-700 active:scale-95 text-red-400">DEL</button>}
          </div>
        ))}
      </div>

      {/* Overlays */}
      {gameState !== 'PLAYING' && (
        <div className="absolute inset-0 bg-black/90 backdrop-blur-md flex flex-col items-center justify-center p-10 text-center animate-in zoom-in">
           <h2 className={`text-5xl font-black mb-4 italic uppercase leading-none ${gameState === 'WON' ? 'text-teal-500' : 'text-red-600'}`}>
             {gameState === 'WON' ? 'CERTIFIED' : 'EXPOSED'}
           </h2>
           <p className="text-zinc-500 text-xs mb-8 uppercase tracking-widest leading-relaxed">
             {gameState === 'WON' ? `You actually belong here. The word was ${targetWord}.` : `You're a poser. Everyone knows the word was ${targetWord}.`}
           </p>
           <button onClick={() => {setGuesses([]); setCurrentGuess(''); setTargetWord(BUZZWORDS[Math.floor(Math.random() * BUZZWORDS.length)]); setGameState('PLAYING')}} className="w-full bg-white text-black py-4 rounded-xl font-black uppercase mb-2">Try Again</button>
           <button onClick={onExit} className="w-full bg-zinc-800 text-zinc-400 py-4 rounded-xl font-black uppercase">Home</button>
        </div>
      )}
    </div>
  );

               <h1 className="text-4xl font-black italic text-white mb-8 uppercase tracking-tighter border-b border-zinc-800 pb-4">APPS</h1>
               <div className="space-y-4">
                 {['egotrip', 'sugarcoat', 'safespot', 'wordlow'].map(id => (
                   <div key={id} className="bg-zinc-900 p-4 rounded-2xl border border-zinc-800 flex items-center gap-4 active:bg-zinc-800 cursor-pointer" onClick={() => setCurrentApp(id)}>
                      <div className="w-12 h-12 bg-zinc-800 rounded-xl flex items-center justify-center shadow-lg">{APPS.find(a => a.id === id).icon}</div>
                      <div>
                        <div className="text-white font-black text-sm uppercase italic">{APPS.find(a => a.id === id).name}</div>
                        <div className="text-[10px] text-zinc-500 font-bold uppercase tracking-widest">Mini-Game</div>
                      </div>
                   </div>
                 ))}
               </div>
               <button onClick={() => setCurrentApp('HOME')} className="mt-auto bg-zinc-900 text-zinc-500 py-4 rounded-2xl font-black uppercase text-xs border border-zinc-800">Exit Games</button>
            </div>
          )}

          {currentApp === 'egotrip' && <EgoTrip onExit={() => setCurrentApp('HOME')} />}
          {currentApp === 'sugarcoat' && <SugarCoat onExit={() => setCurrentApp('HOME')} />}
          {currentApp === 'safespot' && <SafeSpot onExit={() => setCurrentApp('HOME')} />}
          {currentApp === 'wordlow' && <WordLow onExit={() => setCurrentApp('HOME')} />}
        </div>

        <div className="h-10 flex items-center justify-center bg-transparent z-[300]">
          <div className="w-32 h-1.5 bg-zinc-700/50 rounded-full active:bg-white cursor-pointer transition-colors" onClick={() => setCurrentApp('HOME')}></div>
        </div>
      </div>
    </div>
  );
};

export default App;
